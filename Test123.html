// ─── SCHEDULE CALCULATOR ──────────────────────────────────────────────────────

// Returns an array of Date objects, one per reading, spread between start and end.

// If endDate is null, one reading per calendar day (original behaviour).

// If endDate is set, readings are distributed evenly; some days get 2+ readings.

function computeSchedule(plan, startDate, endDate) {

 let total = 0; plan.forEach(e => total += e.days.length);

 const dates = [];


 if (!endDate || endDate <= startDate) {

   // Default: 1 reading per day

   for (let i = 0; i < total; i++) dates.push(addDays(startDate, i));

   return dates;

 }


 const start = new Date(startDate); start.setHours(0,0,0,0);

 const end   = new Date(endDate);   end.setHours(0,0,0,0);

 const totalDays = Math.round((end - start) / 86400000) + 1; // inclusive


 if (totalDays >= total) {

   // Spread evenly — some days get 1, some get 2

   for (let i = 0; i < total; i++) {

     const dayOffset = Math.floor(i * totalDays / total);

     dates.push(addDays(startDate, dayOffset));

   }

 } else {

   // More readings than days — group multiple readings per day

   for (let i = 0; i < total; i++) {

     const dayOffset = Math.min(Math.floor(i * totalDays / total), totalDays - 1);

     dates.push(addDays(startDate, dayOffset));

   }

 }

 return dates;

}


// Returns a human-readable pace string e.g. "~2 readings/day"

function paceLabel(plan, startDate, endDate) {

 if (!endDate || endDate <= startDate) return '';

 let total = 0; plan.forEach(e => total += e.days.length);

 const start = new Date(startDate); start.setHours(0,0,0,0);

 const end   = new Date(endDate);   end.setHours(0,0,0,0);

 const totalDays = Math.max(1, Math.round((end - start) / 86400000) + 1);

 const rate = total / totalDays;

 if (rate <= 1) {

   const daysPerReading = Math.round(totalDays / total);

   return daysPerReading <= 1 ? '1 reading/day' : '1 reading every ' + daysPerReading + ' days';

 }

 return '~' + rate.toFixed(1).replace(/\.0$/,'') + ' readings/day';

}

       else renderNT();

     });

   header.addEventListener('click', function(e) {

     grid.classList.toggle('collapsed', closing);

     header.classList.toggle('collapsed', closing);

   });


   section.appendChild(grid);

 });


 updateProgressBar(checkedObj, plan, fillId, textId);


 if (currentRowEl) setTimeout(() => currentRowEl.scrollIntoView({behavior:'smooth',block:'center'}), 150);

}


function updateProgressBar(checkedObj, plan, fillId, textId, schedule) {

 let total = 0; plan.forEach(e => total += e.days.length);

 const count = Object.keys(checkedObj).filter(k => !!checkedObj[k]).length;

 document.getElementById(fillId).style.width = (count/total*100) + '%';


 // Show "X / Y readings" — if schedule compresses into fewer days, note that

 const isOT = fillId === 'otProgressFill';

 const endDateEl = document.getElementById(isOT ? 'otEndDate' : 'ntEndDate');

 const endVal = endDateEl && endDateEl.value;

 let dayCount = total; // default: 1 reading per day

 if (endVal && schedule && schedule.length) {

   // Count unique dates in the schedule

   const uniqueDates = new Set(schedule.map(d => d.toDateString()));

   dayCount = uniqueDates.size;

 }

 const readingsLabel = count + ' / ' + total + ' readings';

 const daysLabel = dayCount !== total ? ' (' + dayCount + ' days)' : '';

 document.getElementById(textId).textContent = readingsLabel + daysLabel;


 // Determine finish date banner IDs based on which plan this is

 const finishId = isOT ? 'ot-finish-text' : 'nt-finish-text';

 const startDateId = isOT ? 'otStartDate' : 'ntStartDate';


 const startDate = getDate(startDateId);

 const finishEl = document.getElementById(finishId);

 if (!finishEl) return;


 if (count === 0) {

   const endDateEl2 = document.getElementById(isOT ? 'otEndDate' : 'ntEndDate');

   const endVal2 = endDateEl2 && endDateEl2.value;

   let endDisplay;

   if (endVal2) {

     const [ey,em,ed] = endVal2.split('-').map(Number);

     endDisplay = new Date(ey,em-1,ed);

   } else {

     endDisplay = addDays(startDate, total - 1);

   }

   finishEl.innerHTML = 'Target finish: <span class="finish-date">' + endDisplay.toLocaleDateString('en-US',{month:'long',day:'numeric',year:'numeric'}) + '</span>';

   return;

 }


 if (count >= total) {

   finishEl.innerHTML = '🎉 <span class="finish-date">Plan complete!</span> Well done — you have read through the ' + (isOT ? 'Old' : 'New') + ' Testament!';

   return;

 }


 // Find the date of the first unchecked day to estimate pace

 // Days elapsed = index of first unchecked day

 let firstUnchecked = 0;

 let idx = 0;

 for (let si = 0; si < plan.length; si++) {

   for (let di = 0; di < plan[si].days.length; di++) {

     if (!checkedObj['day_'+idx]) { firstUnchecked = idx; break; }

     idx++;

   }

   if (idx !== firstUnchecked || firstUnchecked > 0) break;

 }

 // Find actual first unchecked

 firstUnchecked = 0;

 for (let i = 0; i < total; i++) {

   if (!checkedObj['day_'+i]) { firstUnchecked = i; break; }

 }


 const today = new Date();

 today.setHours(0,0,0,0);

 const start = new Date(startDate);

 start.setHours(0,0,0,0);

 const daysElapsed = Math.max(1, Math.round((today - start) / 86400000));

 const pace = count / daysElapsed; // readings per day

 const remaining = total - count;


 if (pace <= 0) {

   const endDateEl = document.getElementById(isOT ? 'otEndDate' : 'ntEndDate');

 const endDateVal = endDateEl && endDateEl.value;

 let onTrackFinish;

 if (endDateVal) {

   const [ey,em,ed] = endDateVal.split('-').map(Number);

   onTrackFinish = new Date(ey, em-1, ed);

 } else {

   onTrackFinish = addDays(startDate, total - 1);

 }

   finishEl.innerHTML = 'On track to finish: <span class="finish-date">' + onTrackFinish.toLocaleDateString('en-US',{month:'long',day:'numeric',year:'numeric'}) + '</span>';

   return;

 }


 const daysLeft = Math.ceil(remaining / pace);

 const projectedFinish = addDays(today, daysLeft);

 const onTrackFinish = addDays(startDate, total - 1);


 const isAhead = projectedFinish <= onTrackFinish;

 const diffDays = Math.abs(Math.round((projectedFinish - onTrackFinish) / 86400000));


 let statusText = '';

 if (diffDays === 0) {

   statusText = 'You are right on track!';

 } else if (isAhead) {

   statusText = diffDays + ' day' + (diffDays===1?'':'s') + ' ahead of schedule ✨';

 } else {

   statusText = diffDays + ' day' + (diffDays===1?'':'s') + ' behind — keep going!';

 }


 finishEl.innerHTML =

   Math.round(count/total*100) + '% complete &mdash; ' + statusText +

   ' Estimated finish: <span class="finish-date">' +

   projectedFinish.toLocaleDateString('en-US',{month:'long',day:'numeric',year:'numeric'}) +

   '</span>';

}

function jumpToCurrentOT() {

 // Find first unchecked day by scanning otChecked directly

 let total = 0; otPlan.forEach(e => total += e.days.length);

 let targetIdx = -1;

 for (let i = 0; i < total; i++) {

   if (!otChecked['day_'+i]) { targetIdx = i; break; }

 }

 if (targetIdx === -1) return;

 const rows = document.querySelectorAll('#ot-panel .day-row');

 const el = rows[targetIdx];

 if (!el) return;

 const grid = el.closest('.days-grid');

 const header = grid ? grid.previousElementSibling : null;

 if (grid && grid.classList.contains('collapsed')) {

   grid.classList.remove('collapsed');

   grid.style.maxHeight = '9999px';

   if (header) header.classList.remove('collapsed');

 }

 setTimeout(() => el.scrollIntoView({behavior:'smooth', block:'center'}), 50);

}

function getEndDate(id) {

 const val = document.getElementById(id).value;

 if (!val) return null;

 const [y,m,d] = val.split('-').map(Number);

 return new Date(y, m-1, d);

}


function defaultEndDate(startDateId) {

 const start = getDate(startDateId);

 const end = addDays(start, 364); // 365 days inclusive

 return end.toISOString().split('T')[0];

}


function initEndDate(startId, endId) {

 const endEl = document.getElementById(endId);

 if (!endEl.value) endEl.value = defaultEndDate(startId);

 // When start date changes, nudge end date to maintain 365-day default only if user hasn't customized it

 document.getElementById(startId).addEventListener('change', function() {

   // Only auto-update if the gap is still ~365 days (user hasn't customized)

   const currentEnd = getEndDate(endId);

   const currentStart = getDate(startId);

   if (!currentEnd) {

     endEl.value = defaultEndDate(startId);

   } else {

     const gap = Math.round((currentEnd - addDays(currentStart, -1)) / 86400000);

     if (gap === 365) endEl.value = defaultEndDate(startId);

   }

 });

}
